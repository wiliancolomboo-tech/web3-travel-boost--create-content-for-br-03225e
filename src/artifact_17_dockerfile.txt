FROM node:20-slim AS builder
WORKDIR /app
COPY client/package*.json ./
RUN npm ci
COPY client/tsconfig.json ./
COPY client/src ./src
RUN npm run build

FROM node:20-slim
WORKDIR /app
COPY client/package*.json ./
RUN npm ci --only=production
COPY --from=builder /app/dist ./dist
USER node
ENV NODE_ENV=production
CMD ["node", "dist/index.js"]
