import "dotenv/config";
import express from "express";

const app = express();
app.use(express.json());

const seen = new Set<string>();

async function notifyApproved(event: {
  submission: string;
  creator: string;
  contentUrl: string;
}) {
  const key = `approved:${event.submission}`;
  if (seen.has(key)) return;
  seen.add(key);

  const message =
    `Breakpoint Travel Proof approved\n` +
    `Creator: ${event.creator}\n` +
    `Content: ${event.contentUrl}\n` +
    `Submission: ${event.submission}`;

  if (process.env.SLACK_WEBHOOK_URL) {
    const response = await fetch(process.env.SLACK_WEBHOOK_URL, {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ text: message })
    });
    if (!response.ok) throw new Error(`Slack returned ${response.status}`);
  }

  if (process.env.RESEND_API_KEY && process.env.NOTIFY_EMAIL) {
    const response = await fetch("https://api.resend.com/emails", {
      method: "POST",
      headers: {
        authorization: `Bearer ${process.env.RESEND_API_KEY}`,
        "content-type": "application/json"
      },
      body: JSON.stringify({
        from: process.env.RESEND_FROM ?? "travel-proof@example.com",
        to: [process.env.NOTIFY_EMAIL],
        subject: "Breakpoint Travel Proof approved",
        text: message
      })
    });
    if (!response.ok) throw new Error(`Resend returned ${response.status}`);
  }
}

app.get("/healthz", (_req, res) => res.json({ ok: true }));
app.get("/readyz", (_req, res) =>
  res.json({ ok: Boolean(process.env.SOLANA_RPC_URL) })
);

app.post("/events/approved", async (req, res) => {
  try {
    const { submission, creator, contentUrl } = req.body;
    if (
      typeof submission !== "string" ||
      typeof creator !== "string" ||
      typeof contentUrl !== "string"
    ) {
      return res.status(400).json({ error: "invalid event" });
    }

    await notifyApproved({ submission, creator, contentUrl });
    return res.json({ accepted: true });
  } catch (error) {
    return res.status(502).json({
      error: error instanceof Error ? error.message : "notification failed"
    });
  }
});

const port = Number(process.env.PORT ?? 8080);
app.listen(port, "0.0.0.0", () => {
  console.log(`worker listening on ${port}`);
});
