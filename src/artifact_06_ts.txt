import express from "express";
import crypto from "node:crypto";
import { Connection, PublicKey } from "@solana/web3.js";

const RPC_URL = process.env.RPC_URL ?? "https://api.devnet.solana.com";
const PROGRAM_ID = new PublicKey(process.env.PROGRAM_ID!);
const POLL_MS = Number(process.env.POLL_MS ?? 10000);
const PORT = Number(process.env.PORT ?? 8080);

const connection = new Connection(RPC_URL, "confirmed");
const delivered = new Set<string>();

function digest(uri: string): Buffer {
  return crypto.createHash("sha256").update(uri.trim()).digest();
}

async function sendSlack(message: string) {
  const url = process.env.SLACK_WEBHOOK_URL;
  if (!url) return;
  const response = await fetch(url, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({ text: message })
  });
  if (!response.ok) throw new Error(`Slack returned ${response.status}`);
}

async function sendResend(subject: string, html: string) {
  const key = process.env.RESEND_API_KEY;
  const recipient = process.env.REVIEW_EMAIL;
  if (!key || !recipient) return;

  const response = await fetch("https://api.resend.com/emails", {
    method: "POST",
    headers: {
      authorization: `Bearer ${key}`,
      "content-type": "application/json"
    },
    body: JSON.stringify({
      from: process.env.RESEND_FROM ?? "TravelBoost <onboarding@resend.dev>",
      to: [recipient],
      subject,
      html
    })
  });

  if (!response.ok) throw new Error(`Resend returned ${response.status}`);
}

async function poll() {
  const signatures = await connection.getSignaturesForAddress(PROGRAM_ID, {
    limit: 25
  });

  for (const signature of signatures) {
    const key = `signature:${signature.signature}`;
    if (delivered.has(key)) continue;

    const tx = await connection.getParsedTransaction(signature.signature, {
      maxSupportedTransactionVersion: 0
    });

    if (!tx?.meta?.logMessages) continue;

    const isSubmission = tx.meta.logMessages.some((line) =>
      line.includes("Instruction: SubmitContent")
    );

    if (!isSubmission) {
      delivered.add(key);
      continue;
    }

    const message =
      `New TravelBoost submission\n` +
      `Signature: ${signature.signature}\n` +
      `Explorer: https://explorer.solana.com/tx/${signature.signature}?cluster=devnet`;

    await sendSlack(message);
    await sendResend(
      "TravelBoost: new Breakpoint content submission",
      `<p>${message.replaceAll("\n", "<br>")}</p>`
    );

    delivered.add(key);
  }
}

const app = express();

app.get("/healthz", async (_req, res) => {
  try {
    const slot = await connection.getSlot();
    res.json({
      ok: true,
      rpc: RPC_URL,
      slot,
      program: PROGRAM_ID.toBase58()
    });
  } catch {
    res.status(503).json({ ok: false });
  }
});

app.listen(PORT, () => {
  console.log(`TravelBoost worker listening on :${PORT}`);
});

setInterval(() => {
  poll().catch((error) => console.error("poll failed", error));
}, POLL_MS);

poll().catch((error) => console.error("initial poll failed", error));

export { digest };
