FROM node:20-alpine AS builder

WORKDIR /usr/src/app

COPY oracle-worker/package*.json ./
RUN npm ci

COPY oracle-worker/tsconfig.json ./
COPY oracle-worker/src ./src
COPY target/idl/breakpoint_travel_boost.json ./src/idl.json

RUN npm run build

FROM node:20-alpine AS runner
WORKDIR /usr/src/app

ENV NODE_ENV=production

COPY oracle-worker/package*.json ./
RUN npm ci --only=production

COPY --from=builder /usr/src/app/dist ./dist
COPY target/idl/breakpoint_travel_boost.json ./dist/idl.json

EXPOSE 8080
USER node

CMD ["node", "dist/index.js"]
